# --- Power protection -------------------------------------------------------
# Witty Pi asks the Pi to shut down cleanly when Vin drops below LOW_VOLTAGE.
# 11.5 V suits a 12 V lead-acid/AGM battery under light load (~20% charge left).
# For a 12 V (4S) LiFePO4 battery use 12.0 instead.  0 = disabled.
LOW_VOLTAGE 11.0

# Power the Pi back on automatically once Vin rises above this (e.g. solar
# recharge).  0 = disabled: the Pi then just waits for the next ON window.
RECOVERY_VOLTAGE 0

# Seconds after power is (re)connected before the Pi is switched on.
# Lets the Pi restart after a total power loss.  255 = never auto-start.
DEFAULT_ON_DELAY 10
